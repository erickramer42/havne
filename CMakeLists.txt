# Havne version is the single source of truth; propagate to core + app
configure_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/core/include/ecuio/version.h.in
    ${CMAKE_CURRENT_BINARY_DIR}/generated/ecuio/version.h
    @ONLY
)

# The surrogate and its client compile the same two headers in separate
# trees (x86 and x64), and a constant changed in one tree only would
# pass over the pipe unnoticed. Both get a fingerprint of the headers'
# bytes and compare it at OP_OPEN. Editing either header re-runs this
# configure step, so the fingerprint follows the source.
set(_app_shared_headers
    ${CMAKE_CURRENT_SOURCE_DIR}/core/include/ecuio/j2534_ptypes.h
    ${CMAKE_CURRENT_SOURCE_DIR}/core/include/ecuio/surrogate_protocol.h
)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_app_shared_headers})
set(_app_shared_sums "")
foreach(_app_header IN LISTS _app_shared_headers)
    file(SHA256 ${_app_header} _app_sum)
    string(APPEND _app_shared_sums ${_app_sum})
endforeach()
string(SHA256 _app_sum "${_app_shared_sums}")
string(SUBSTRING ${_app_sum} 0 8 ECUIO_SUR_FINGERPRINT_HEX)
file(CONFIGURE
    OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/ecuio/sur_fingerprint.h
    CONTENT [[/* Generated by CMake: the first 32 bits of a SHA-256 over
 * j2534_ptypes.h and surrogate_protocol.h. The surrogate client and the
 * child compare it at OP_OPEN (surrogate_protocol.h). */
#ifndef ECUIO_SUR_FINGERPRINT_H
#define ECUIO_SUR_FINGERPRINT_H
#define ECUIO_SUR_FINGERPRINT 0x@ECUIO_SUR_FINGERPRINT_HEX@u
#endif
]]
    @ONLY
)

# QTTUNE_BUILD_CORE_ONLY is the option's name before the rename (ADR-0001's
# appendix of 2026-10-28), honored until v0.3.4 so a configured tree or a
# script that passes it still builds core alone
if(DEFINED QTTUNE_BUILD_CORE_ONLY AND NOT DEFINED HAVNE_BUILD_CORE_ONLY)
    message(WARNING "QTTUNE_BUILD_CORE_ONLY is now HAVNE_BUILD_CORE_ONLY; "
        "reconfigure with -DHAVNE_BUILD_CORE_ONLY=${QTTUNE_BUILD_CORE_ONLY}")
    set(HAVNE_BUILD_CORE_ONLY ${QTTUNE_BUILD_CORE_ONLY} CACHE BOOL
        "Build core library and tests; skip Qt app")
endif()
option(HAVNE_BUILD_CORE_ONLY "Build core library and tests; skip Qt app" OFF)

# libFuzzer harnesses (fuzz/): Clang or MSVC, off by default, in a tree
# of their own, never in ctest. The CI 'fuzz' job builds and runs them;
# fuzz/README.md has the local recipe.
option(ECUIO_BUILD_FUZZERS "Build libFuzzer harnesses (Clang or MSVC)" OFF)

# Compiler warnings for every first-party target: core, the surrogate,
# the app, the tests and the fake DLLs. They don't fail the build (no /WX
# or -Werror): CI compiles with GCC, MSVC and AppleClang, and one
# compiler's new warning shouldn't break the others' builds.
function(app_enable_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /guard:cf)
        # /GS is on by default in MSVC
    else()
        target_compile_options(${target} PRIVATE
            -Wall -Wextra -fstack-protector-strong)
        # _FORTIFY_SOURCE only works with optimization; ASan builds
        # skip it (redundant with the sanitizer and warns at -O0)
        get_target_property(_app_san ${target} CXX_COMPILE_OPTIONS)
        if(NOT (_app_san MATCHES "sanitize"))
            target_compile_options(${target} PRIVATE -D_FORTIFY_SOURCE=2)
        endif()
        # Full RELRO is an ELF linker option. Apple's ld has no -z (the
        # link fails with "unknown options: -z"), and Mach-O already
        # makes __DATA_CONST read-only after fixups.
        if(NOT APPLE)
            target_link_options(${target} PRIVATE
                -Wl,-z,relro -Wl,-z,now)
        endif()
    endif()
endfunction()

# Windows: a VERSIONINFO resource for a shipped executable, from
# project(Havne VERSION ...): Explorer's Details tab shows it, code
# signing reads it, and stage.ps1 checks it against the release.
# <name> is the file's name without .exe. An optional fourth argument
# names an .ico, which becomes the executable's icon (IDI_ICON1, which
# Qt's Windows plugin also takes for the window's).
set(APP_VERSION_RC_IN "${CMAKE_CURRENT_SOURCE_DIR}/packaging/windows/version.rc.in")
function(app_add_version_resource target name description)
    if(NOT WIN32)
        return()
    endif()
    set(APP_RC_NAME "${name}")
    set(APP_RC_DESCRIPTION "${description}")
    set(APP_RC_ICON "")
    if(ARGC GREATER 3)
        set(APP_RC_ICON "IDI_ICON1 ICON \"${ARGV3}\"")
    endif()
    set(rc "${CMAKE_CURRENT_BINARY_DIR}/${name}_version.rc")
    configure_file("${APP_VERSION_RC_IN}" "${rc}" @ONLY)
    target_sources(${target} PRIVATE "${rc}")
endfunction()

# Core never links Qt (ADR-0001). This runs after every subdirectory, so
# it sees links added from anywhere. It reads core's link lists, because
# Qt's imported targets are scoped to the directory that found them and a
# TARGET Qt6::Core test can't see them from here.
foreach(_app_prop LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
    get_target_property(_app_libs ecuio-core ${_app_prop})
    if(_app_libs AND _app_libs MATCHES "Qt[0-9]*::")
        message(FATAL_ERROR
            "ecuio-core must not link Qt (${_app_prop}: ${_app_libs})")
    endif()
endforeach()
