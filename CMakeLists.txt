# root/CMakeLists.txt
cmake_minimum_required(VERSION 3.21)

project(QtTune
    VERSION 0.3.0
    DESCRIPTION "Cross-platform ECU reading, logging, and tuning"
    LANGUAGES C CXX
)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)

# QtTune version is the single source of truth; propagate to core + app
configure_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/core/include/ecuio/version.h.in
    ${CMAKE_CURRENT_BINARY_DIR}/generated/ecuio/version.h
    @ONLY
)

# The surrogate and its client compile the same two headers in separate
# trees (x86 and x64), and a constant changed in one tree only would
# pass over the pipe unnoticed. Both get a fingerprint of the headers'
# bytes and compare it at OP_OPEN. Editing either header re-runs this
# configure step, so the fingerprint follows the source.
set(_qttune_shared_headers
    ${CMAKE_CURRENT_SOURCE_DIR}/core/include/ecuio/j2534_ptypes.h
    ${CMAKE_CURRENT_SOURCE_DIR}/core/include/ecuio/surrogate_protocol.h
)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_qttune_shared_headers})
set(_qttune_shared_sums "")
foreach(_qttune_header IN LISTS _qttune_shared_headers)
    file(SHA256 ${_qttune_header} _qttune_sum)
    string(APPEND _qttune_shared_sums ${_qttune_sum})
endforeach()
string(SHA256 _qttune_sum "${_qttune_shared_sums}")
string(SUBSTRING ${_qttune_sum} 0 8 ECUIO_SUR_FINGERPRINT_HEX)
file(CONFIGURE
    OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/ecuio/sur_fingerprint.h
    CONTENT [[/* Generated by CMake: the first 32 bits of a SHA-256 over
 * j2534_ptypes.h and surrogate_protocol.h. The surrogate client and the
 * child compare it at OP_OPEN (surrogate_protocol.h). */
#ifndef ECUIO_SUR_FINGERPRINT_H
#define ECUIO_SUR_FINGERPRINT_H
#define ECUIO_SUR_FINGERPRINT 0x@ECUIO_SUR_FINGERPRINT_HEX@u
#endif
]]
    @ONLY
)

option(QTTUNE_BUILD_CORE_ONLY "Build core library and tests; skip Qt app" OFF)

# libFuzzer harnesses (fuzz/): Clang or MSVC, off by default, in a tree
# of their own, never in ctest. The CI 'fuzz' job builds and runs them;
# fuzz/README.md has the local recipe.
option(ECUIO_BUILD_FUZZERS "Build libFuzzer harnesses (Clang or MSVC)" OFF)

# Compiler warnings for every first-party target: core, the surrogate,
# the app, the tests and the fake DLLs. They don't fail the build (no /WX
# or -Werror): CI compiles with GCC, MSVC and AppleClang, and one
# compiler's new warning shouldn't break the others' builds.
function(qttune_enable_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra)
    endif()
endfunction()

add_subdirectory(core)
add_subdirectory(surrogate)

include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()

if(ECUIO_BUILD_FUZZERS)
    add_subdirectory(fuzz)
endif()

if(NOT QTTUNE_BUILD_CORE_ONLY)
    add_subdirectory(app)
endif()

# Core never links Qt (ADR-0001). This runs after every subdirectory, so
# it sees links added from anywhere. It reads core's link lists, because
# Qt's imported targets are scoped to the directory that found them and a
# TARGET Qt6::Core test can't see them from here.
foreach(_qttune_prop LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
    get_target_property(_qttune_libs ecuio-core ${_qttune_prop})
    if(_qttune_libs AND _qttune_libs MATCHES "Qt[0-9]*::")
        message(FATAL_ERROR
            "ecuio-core must not link Qt (${_qttune_prop}: ${_qttune_libs})")
    endif()
endforeach()
